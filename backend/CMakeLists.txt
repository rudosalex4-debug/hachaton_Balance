cmake_minimum_required(VERSION 3.22)
project(decision_lens LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_SERVER "Build the Linux HTTP server and PostgreSQL storage" ON)
find_package(jsoncpp CONFIG REQUIRED)
find_package(simdjson CONFIG REQUIRED)
add_library(lens_core src/core.cpp)
target_include_directories(lens_core PUBLIC include)
target_link_libraries(lens_core PUBLIC JsonCpp::JsonCpp simdjson::simdjson)
add_library(lens_reports src/reports.cpp)
target_include_directories(lens_reports PUBLIC include)
target_link_libraries(lens_reports PUBLIC JsonCpp::JsonCpp)
if(BUILD_SERVER)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build the server on Linux/WSL2 with g++, or in Docker Compose.")
 endif()
 find_package(Drogon CONFIG REQUIRED)
 find_package(OpenSSL REQUIRED)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(PQXX REQUIRED IMPORTED_TARGET libpqxx>=7.8)
 add_executable(decision_lens src/server.cpp src/storage.cpp src/stage_executor.cpp)
 target_include_directories(decision_lens PRIVATE include)
 target_link_libraries(decision_lens PRIVATE lens_core lens_reports Drogon::Drogon PkgConfig::PQXX OpenSSL::Crypto)
 include(CheckCXXSourceCompiles)
 set(CMAKE_REQUIRED_LIBRARIES PkgConfig::PQXX)
 check_cxx_source_compiles("#include <pqxx/pqxx>\nint main(){throw pqxx::conversion_overrun{\"abi probe\"};}" LENS_PQXX_CXX20_EXCEPTION_ABI)
 unset(CMAKE_REQUIRED_LIBRARIES)
 if(NOT LENS_PQXX_CXX20_EXCEPTION_ABI)
  target_compile_definitions(decision_lens PRIVATE LENS_PQXX_LEGACY_EXCEPTION_ABI)
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(GTest REQUIRED)
 add_executable(core_tests tests/core_test.cpp tests/report_test.cpp)
 target_link_libraries(core_tests PRIVATE lens_core lens_reports GTest::gtest_main)
 target_compile_definitions(core_tests PRIVATE LENS_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/fixtures")
 include(GoogleTest)
 gtest_discover_tests(core_tests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark.cpp")
 add_executable(core_benchmark tests/benchmark.cpp)
 target_link_libraries(core_benchmark PRIVATE lens_core)
endif()
